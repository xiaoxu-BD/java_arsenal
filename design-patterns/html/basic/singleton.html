核心规则（考试就考这三条）
  • 私有构造器：堵死外界 new 的入口，实例的"出生"只能由类自己控制
  • 静态获取入口：getInstance() 是拿实例的唯一通道，内部保证只造一次
  • 构造器执行次数 = 1：无论多少次调用、多少线程并发，身份号永远相同
饿汉式：简单但不管用不用都占内存｜懒汉式 DCL：第一次使用才创建，volatile 防止指令重排拿到"半成品"。 高级版见 advanced 包：静态内部类 Holder 与枚举单例，还能防反射/反序列化攻击。
代码联动 · 操作时对应行会高亮
public class EagerSingleton {
    private static final EagerSingleton INSTANCE = new EagerSingleton();

    private EagerSingleton() {          // 私有构造：外界无法 new
    }

    public static EagerSingleton getInstance() {
        return INSTANCE;                // 永远返回同一个
    }
}
public class LazySingleton {
    private static volatile LazySingleton instance;

    private LazySingleton() {           // 私有构造：外界无法 new
    }

    public static LazySingleton getInstance() {
        if (instance == null) {              // ① 一检：无锁快路径
            synchronized (LazySingleton.class) {
                if (instance == null) {      // ② 二检：防等锁线程重复创建
                    instance = new LazySingleton();  // ③ 真正创建
                }
            }
        }
        return instance;               // ④ 永远返回同一个
    }
}
